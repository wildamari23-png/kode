const SS = SpreadsheetApp.getActiveSpreadsheet();

function createSheetIfNotExists(sheetName, headers) {
  let sheet = SS.getSheetByName(sheetName);
  if (!sheet) {
    sheet = SS.insertSheet(sheetName);
    sheet.appendRow(headers);
    sheet.setFrozenRows(1);
    sheet.getRange(1, 1, 1, headers.length).setFontWeight("bold").setHorizontalAlignment("center");
  }
  return sheet;
}

const SHEETS = {
  USERS: SS.getSheetByName("Users"),
  ORDER: SS.getSheetByName("Order"),
  LAYANAN: SS.getSheetByName("Layanan"),
  PENGELUARAN: SS.getSheetByName("Pengeluaran") || createSheetIfNotExists("Pengeluaran", ["ID Pengeluaran", "Tanggal", "Deskripsi", "Jumlah", "Kategori", "Input By", "Timestamp", "Sumber_dana"]),
  GAJI: SS.getSheetByName("Gaji") || createSheetIfNotExists("Gaji", ["ID Gaji", "Worker Username", "Bulan/Tahun", "Gaji Pokok", "Bonus", "Potongan", "Total Terima", "Input By", "Timestamp"])
};

function getSheetData(sheetName) {
  const sheet = SS.getSheetByName(sheetName);
  if (!sheet) return [];
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];
  const headers = data[0];
  return data.slice(1).map(row => {
    const obj = {};
    headers.forEach((h, i) => {
      let val = row[i];
      if (val instanceof Date) val = val.toISOString();
      obj[String(h).trim()] = val; // Trim to avoid spaces issues
    });
    return obj;
  });
}

function getPublicServices() { 
  const all = getSheetData("Layanan");
  
  return all.filter(s => {
    const status = String(s['Status Layanan'] || '').trim();
    
    // HANYA yang statusnya persis "Tersedia"
    return status === "Tersedia" || 
           status.toLowerCase() === "tersedia";
  });
}

function getAllServices() { 
  return getSheetData("Layanan"); 
}

function getDashboardData(userRole, username) {
  const orders = getSheetData("Order");
  const services = getAllServices();   
  const users = getSheetData("Users");
  let filtered = orders;
  const roleStr = String(userRole).toLowerCase();

  // Only filter main monitor list if worker. Bendahara and Admin see all.
  if (roleStr === "worker") {
    filtered = orders.filter(o => !o.WORKER || o.WORKER === username);
  }

  return { 
    orders: filtered.reverse(), 
    allServices: services, 
    users: users, 
    runningText: getRunningText(),
    pengeluaran: getSheetData("Pengeluaran").reverse(), 
    gaji: getSheetData("Gaji").reverse() 
  };
}

function login(username, password) {
  const users = getSheetData("Users");
  const user = users.find(u => u.Username === username && String(u.Password) === String(password));
  if (user) { return { success: true, user: user }; }
  return { success: false, message: "Username atau Password salah!" };
}

function updateWorkerAccount(payload) {
  const sheet = SHEETS.USERS;
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === payload.id) {
      const row = i + 1;
      sheet.getRange(row, 5).setValue(payload.name);
      if (payload.password && payload.password.trim() !== "") sheet.getRange(row, 3).setValue(payload.password);
      sheet.getRange(row, 6).setValue(payload.bank); 
      sheet.getRange(row, 7).setValue(payload.rekening);
      sheet.getRange(row, 9).setValue(payload.email);
      return { success: true };
    }
  }
  return { success: false, message: "User tidak ditemukan" };
}

function saveOrder(payload) {
  try {
    const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Order");
    if (!sheet) return { success: false, message: "Sheet 'Order' tidak ditemukan" };

    const headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0];

    // Mapping kolom yang EXACT sesuai sheet Anda
    let col = {};
    headers.forEach((h, i) => {
      const key = String(h).trim();
      if (key === "ID PESANAN") col.id = i;
      else if (key === "NAMA PELANGGAN") col.nama = i;
      else if (key === "NO HANDPHONE") col.hp = i;
      else if (key === "EMAIL") col.email = i;
      else if (key === "LAYANAN") col.layanan = i;
      else if (key === "RINCIAN PESANAN") col.rincian = i;
      else if (key === "QTY") col.qty = i;
      else if (key === "SATUAN") col.satuan = i;                    // Kolom H
      else if (key === "HARGA SATUAN") col.hargaSatuan = i;         // Kolom I
      else if (key === "TOTAL HARGA") col.totalHarga = i;
      else if (key === "TAHAPAN PESANAN") col.tahapan = i;
      else if (key === "STATUS PEMBAYARAN") col.statusBayar = i;
      else if (key === "JENIS PEMBAYARAN") col.jenisBayar = i;
      else if (key === "WORKER") col.worker = i;                    // Kolom P
      else if (key === "PENGINPUT PESANAN") col.penginput = i;      // Kolom S
      else if (key === "WAKTU  INPUT PESANAN" || key === "WAKTU INPUT PESANAN") col.waktuInput = i; // Kolom T
      else if (key === "WAKTU DITERIMA WORKER") col.waktuTerima = i; // Kolom U
      else if (key === "DISCOUNT") col.discount = i;
      else if (key === "DP") col.dp = i;
      else if (key === "Harga_display") col.hargaDisplay = i;
      else if (key === "fee admin") col.feeAdmin = i;
      else if (key === "fee perusahaan") col.feePerusahaan = i;
      else if (key === "DEADLINE") col.deadline = i;
    });

    const newId = "LC-" + Utilities.getUuid().substring(0, 5).toUpperCase();
    const row = new Array(headers.length).fill("");

    // Isi data dasar dari form
    if (col.id !== undefined) row[col.id] = newId;
    if (col.nama !== undefined) row[col.nama] = payload["NAMA PELANGGAN"] || "";
    if (col.hp !== undefined) row[col.hp] = payload["NO HANDPHONE"] || "";
    if (col.email !== undefined) row[col.email] = payload["EMAIL"] || "";
    if (col.layanan !== undefined) row[col.layanan] = payload["LAYANAN"] || "";
    if (col.rincian !== undefined) row[col.rincian] = payload["RINCIAN PESANAN"] || "";
    if (col.qty !== undefined) row[col.qty] = Number(payload["QTY"]) || 1;
    if (col.totalHarga !== undefined) row[col.totalHarga] = Number(payload["TOTAL HARGA"]) || 0;

    // AUTO-FILL DARI SHEET LAYANAN
    const layananName = String(payload["LAYANAN"] || "").trim();
    if (layananName) {
      const services = getAllServices();
      const svc = services.find(s => 
        String(s['Detail Layanan'] || s['ID LAYANAN']).trim() === layananName
      );
      if (svc) {
        if (col.satuan !== undefined) row[col.satuan] = svc.Satuan || "Unit";                    // Kolom H
        if (col.hargaDisplay !== undefined) row[col.hargaDisplay] = Number(svc['Harga Publish']) || 0;
        if (col.feeAdmin !== undefined) row[col.feeAdmin] = Number(svc['fee Admin(%)']) || 15;
        if (col.feePerusahaan !== undefined) row[col.feePerusahaan] = Number(svc['fee perusahaan(%)']) || 5;
      }
    }

    // AUTO PENGINPUT PESANAN (Kolom S)
    if (col.penginput !== undefined) {
      row[col.penginput] = payload.penginput || "ADMIN";
    }

    // Data otomatis
    if (col.tahapan !== undefined) row[col.tahapan] = "Antrean";
    if (col.statusBayar !== undefined) row[col.statusBayar] = payload["STATUS PEMBAYARAN"] || "Unpaid Invoice/Bill";
    if (col.jenisBayar !== undefined) row[col.jenisBayar] = payload["JENIS PEMBAYARAN"] || "QRIS";
    if (col.worker !== undefined) row[col.worker] = payload["WORKER"] || "";
    if (col.waktuInput !== undefined) row[col.waktuInput] = new Date();   // Kolom T
    if (col.discount !== undefined) row[col.discount] = Number(payload["DISCOUNT"]) || 0;
    if (col.dp !== undefined) row[col.dp] = Number(payload["DP"]) || 0;
    if (col.deadline !== undefined) row[col.deadline] = payload["DEADLINE"] || "";

    sheet.appendRow(row);

    // Otomatis kirim template WA DP jika pesanan baru langsung berstatus DP
    const statusBayarBaru = String(payload["STATUS PEMBAYARAN"] || '').trim();
    if (statusBayarBaru === "Down Payment (DP)") {
      try { sendPaymentTemplateWA(newId, "DP"); } catch (e) { console.log("Auto WA DP gagal:", e); }
    }

    return { success: true, id: newId, message: "Pesanan berhasil disimpan" };

  } catch (e) {
    console.error("Error saveOrder:", e);
    return { success: false, message: e.message };
  }
}

function updateOrder(orderId, updates) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("Order");
    if (!sheet) return { success: false, message: "Sheet Order tidak ditemukan" };

    const data = sheet.getDataRange().getValues();
    const headers = data[0];

    // Mapping kolom
    let idCol = -1;
    let statusCol = -1;
    let workerCol = -1;
    let waktuTerimaCol = -1;   // Kolom U = WAKTU DITERIMA WORKER
    let waktuSelesaiCol = -1;  // Kolom baru = WAKTU SELESAI
    let paymentCol = -1;
    let finalFolderCol = -1;
    let initialFolderCol = -1;

    for (let i = 0; i < headers.length; i++) {
      const h = String(headers[i]).trim().toUpperCase();
      if (h === "ID PESANAN") idCol = i;
      else if (h === "TAHAPAN PESANAN") statusCol = i;
      else if (h === "WORKER") workerCol = i;
      else if (h === "WAKTU DITERIMA WORKER" || h.includes("WAKTU DITERIMA")) waktuTerimaCol = i;
      else if (h === "WAKTU SELESAI" || h.includes("WAKTU SELESAI")) waktuSelesaiCol = i;
      else if (h === "STATUS PEMBAYARAN") paymentCol = i;
      else if (h === "FOLDER DOKUMEN FINAL") finalFolderCol = i;
      else if (h === "FOLDER DOKUMEN AWAL") initialFolderCol = i;
    }

    if (idCol === -1) return { success: false, message: "Kolom ID PESANAN tidak ditemukan" };

    // Cari baris order
    let rowIndex = -1;
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][idCol]).trim() === String(orderId).trim()) {
        rowIndex = i + 1;
        break;
      }
    }
    if (rowIndex === -1) return { success: false, message: "Order tidak ditemukan" };

    // Update Status
    const previousPayment = paymentCol !== -1 ? String(data[rowIndex - 1][paymentCol] || '').trim() : '';
    if (updates.status && statusCol !== -1) {
      sheet.getRange(rowIndex, statusCol + 1).setValue(updates.status);
    }
    if (updates.payment && paymentCol !== -1) {
      sheet.getRange(rowIndex, paymentCol + 1).setValue(updates.payment);
    }

    // Update Worker (Kolom P)
    if (updates.worker && workerCol !== -1) {
      sheet.getRange(rowIndex, workerCol + 1).setValue(updates.worker);
    }

    // ==================== OTOMATIS ISI KOLOM U ====================
    if (updates.status === "Proses" && waktuTerimaCol !== -1) {
      const currentValue = sheet.getRange(rowIndex, waktuTerimaCol + 1).getValue();
      // Hanya isi jika kolom U masih kosong (agar tidak menimpa timestamp lama)
      if (!currentValue) {
        sheet.getRange(rowIndex, waktuTerimaCol + 1).setValue(new Date());
      }
    }

    // Simpan timestamp selesai agar durasi berhenti ketika status Selesai
    if (updates.status === "Selesai") {
      if (waktuSelesaiCol === -1) {
        waktuSelesaiCol = headers.length;
        sheet.getRange(1, waktuSelesaiCol + 1).setValue("WAKTU SELESAI");
      }
      const finishCell = sheet.getRange(rowIndex, waktuSelesaiCol + 1);
      if (!finishCell.getValue()) finishCell.setValue(new Date());
    }

    // Update Catatan Worker (Kolom R)
    if (updates.note) {
      let noteCol = headers.indexOf("CATATAN WORKER");
      if (noteCol !== -1) {
        sheet.getRange(rowIndex, noteCol + 1).setValue(updates.note);
      }
    }

    if (updates.finalFiles && updates.finalFiles.length > 0) {
      const folderUrl = finalFolderCol !== -1 ? String(sheet.getRange(rowIndex, finalFolderCol + 1).getValue() || '') : '';
      const orderFolderUrl = initialFolderCol !== -1 ? String(sheet.getRange(rowIndex, initialFolderCol + 1).getValue() || '') : '';
      const folder = getOrCreateOrderFolder(orderId, 'final', folderUrl, orderFolderUrl);
      updates.finalFiles.forEach(f => {
        const blob = Utilities.newBlob(Utilities.base64Decode(f.data), f.mimeType || 'application/octet-stream', f.filename || ('file-' + Date.now()));
        folder.createFile(blob);
      });
      if (finalFolderCol !== -1 && !sheet.getRange(rowIndex, finalFolderCol + 1).getValue()) {
        sheet.getRange(rowIndex, finalFolderCol + 1).setValue(folder.getUrl());
      }
    }

    // Otomatis kirim WA pembayaran saat transisi status bayar
    if (updates.payment) {
      const newPayment = String(updates.payment || '').trim();
      if (newPayment === "Paid Full" && previousPayment !== "Paid Full") {
        try { sendPaymentTemplateWA(orderId, "LUNAS"); } catch (e) { console.log("Auto WA Lunas gagal:", e); }
      } else if (newPayment === "Down Payment (DP)" && previousPayment !== "Down Payment (DP)") {
        try { sendPaymentTemplateWA(orderId, "DP"); } catch (e) { console.log("Auto WA DP gagal:", e); }
      }
    }

    // Alur baru: kirim template penagihan pelunasan saat status masuk penagihan pembayaran akhir
    if (updates.status && String(updates.status).trim() === "Penagihan Pembayaran Akhir") {
      try { sendPaymentTemplateWA(orderId, "LUNAS"); } catch (e) { console.log("Auto WA Penagihan Akhir gagal:", e); }
    }

    return { success: true, message: "Order berhasil diupdate" };

  } catch (e) {
    console.error("Error updateOrder:", e);
    return { success: false, message: "Gagal mengupdate order" };
  }
}

function getOrderRowAndHeaders(orderId) {
  const sheet = SHEETS.ORDER;
  const data = sheet.getDataRange().getValues();
  const headers = data[0].map(h => String(h).trim());
  const idCol = headers.findIndex(h => h.toUpperCase() === 'ID PESANAN');
  if (idCol === -1) throw new Error('Kolom ID PESANAN tidak ditemukan');
  for (let i = 1; i < data.length; i++) {
    if (String(data[i][idCol]).trim() === String(orderId).trim()) {
      return { sheet, headers, rowIndex: i + 1, row: data[i] };
    }
  }
  throw new Error('Order tidak ditemukan');
}

function extractDriveFolderId(urlOrId) {
  const value = String(urlOrId || '').trim();
  if (!value) return '';
  const folderMatch = value.match(/\/folders\/([-\w]+)/);
  if (folderMatch && folderMatch[1]) return folderMatch[1];
  const m = value.match(/[-\w]{20,}/);
  return m ? m[0] : value;
}

function getOrCreateOrderFolder(orderId, type, currentFolderUrl, orderFolderUrl) {
  const rootFolderId = '11XCjzCeG4HtojpUUKMrObkb-aefkgkfr';
  const root = DriveApp.getFolderById(rootFolderId);

  if (currentFolderUrl) {
    const existingId = extractDriveFolderId(currentFolderUrl);
    if (existingId) {
      try { return DriveApp.getFolderById(existingId); } catch (e) {}
    }
  }

  let orderFolder = null;
  const orderFolderId = extractDriveFolderId(orderFolderUrl);
  if (orderFolderId) {
    try { orderFolder = DriveApp.getFolderById(orderFolderId); } catch (e) {}
  }
  if (!orderFolder) {
    const matches = root.getFoldersByName(orderId);
    orderFolder = matches.hasNext() ? matches.next() : root.createFolder(orderId);
  }

  const subName = type === 'final' ? 'DOKUMEN_FINAL' : 'DOKUMEN_AWAL';
  const sub = orderFolder.getFoldersByName(subName);
  return sub.hasNext() ? sub.next() : orderFolder.createFolder(subName);
}

function getOrderDocumentFiles(orderId, docType) {
  try {
    const { headers, row } = getOrderRowAndHeaders(orderId);
    const isFinal = String(docType || '').toLowerCase() === 'final';
    const folderKey = isFinal ? 'FOLDER DOKUMEN FINAL' : 'FOLDER DOKUMEN AWAL';
    const folderCol = headers.findIndex(h => h.toUpperCase() === folderKey);
    const orderFolderCol = headers.findIndex(h => h.toUpperCase() === 'FOLDER DOKUMEN AWAL');
    const folderUrl = folderCol !== -1 ? String(row[folderCol] || '') : '';
    const orderFolderUrl = orderFolderCol !== -1 ? String(row[orderFolderCol] || '') : '';
    const folder = getOrCreateOrderFolder(orderId, isFinal ? 'final' : 'initial', folderUrl, orderFolderUrl);

    const files = [];
    const it = folder.getFiles();
    while (it.hasNext()) {
      const f = it.next();
      files.push({
        id: f.getId(),
        name: f.getName(),
        mimeType: f.getMimeType(),
        url: f.getUrl(),
        reviewUrl: `https://drive.google.com/file/d/${f.getId()}/preview`,
        downloadUrl: `https://drive.google.com/uc?export=download&id=${f.getId()}`,
        sizeKb: Math.round((f.getSize() || 0) / 1024),
        updatedAt: Utilities.formatDate(f.getLastUpdated(), Session.getScriptTimeZone() || 'GMT+7', 'dd/MM/yyyy HH:mm')
      });
    }
    return { success: true, folderUrl: folder.getUrl(), files };
  } catch (e) {
    return { success: false, message: e.toString(), files: [] };
  }
}

function uploadOrderDocumentFiles(orderId, docType, files) {
  try {
    const ctx = getOrderRowAndHeaders(orderId);
    const isFinal = String(docType || '').toLowerCase() === 'final';
    const folderKey = isFinal ? 'FOLDER DOKUMEN FINAL' : 'FOLDER DOKUMEN AWAL';
    const folderCol = ctx.headers.findIndex(h => h.toUpperCase() === folderKey);
    const orderFolderCol = ctx.headers.findIndex(h => h.toUpperCase() === 'FOLDER DOKUMEN AWAL');
    const folderUrl = folderCol !== -1 ? String(ctx.row[folderCol] || '') : '';
    const orderFolderUrl = orderFolderCol !== -1 ? String(ctx.row[orderFolderCol] || '') : '';

    const folder = getOrCreateOrderFolder(orderId, isFinal ? 'final' : 'initial', folderUrl, orderFolderUrl);
    (files || []).forEach(f => {
      const blob = Utilities.newBlob(Utilities.base64Decode(f.data), f.mimeType || 'application/octet-stream', f.filename || ('upload-' + Date.now()));
      folder.createFile(blob);
    });

    if (folderCol !== -1) ctx.sheet.getRange(ctx.rowIndex, folderCol + 1).setValue(folder.getUrl());
    return { success: true, message: 'File berhasil diupload', folderUrl: folder.getUrl() };
  } catch (e) {
    return { success: false, message: e.toString() };
  }
}

function downloadOrderDocumentZip(orderId, docType) {
  try {
    const data = getOrderDocumentFiles(orderId, docType);
    if (!data.success) return data;
    if (!data.files || data.files.length === 0) return { success: false, message: 'Belum ada file untuk di-zip.' };
    const blobs = data.files.map(f => DriveApp.getFileById(f.id).getBlob().setName(f.name));
    const zipBlob = Utilities.zip(blobs, `${orderId}-${docType || 'dokumen'}.zip`);
    const file = DriveApp.createFile(zipBlob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    return { success: true, url: file.getDownloadUrl(), fileUrl: file.getUrl(), name: file.getName() };
  } catch (e) {
    return { success: false, message: e.toString() };
  }
}

function deleteOrder(id) {
  const sheet = SHEETS.ORDER;
  const data = sheet.getDataRange().getValues();
  for (let i = 1; i < data.length; i++) {
    if (data[i][0] === id) { sheet.deleteRow(i + 1); return { success: true }; }
  }
  return { success: false, message: "ID tidak ditemukan" };
}

function calculateOrderFinanceServer(o) {
  const gross = Number(o['TOTAL HARGA']) || 0;
  const discountP = Number(o.DISCOUNT) || 0;
  const net = gross - (gross * discountP / 100);
  const dpP = Number(o.DP) || 0;
  const originalDpAmount = net * dpP / 100;
  const isPaidFull = String(o['STATUS PEMBAYARAN'] || '').toLowerCase() === 'paid full'
    || String(o['TAHAPAN PESANAN'] || '').toLowerCase() === 'selesai';
  const dpAmount = isPaidFull ? 0 : originalDpAmount;
  const balance = isPaidFull ? 0 : (net - dpAmount);
  const pelunasanAmount = isPaidFull ? (net - originalDpAmount) : 0;
  return { gross, discountP, net, dpP, dpAmount, balance, isPaidFull, originalDpAmount, pelunasanAmount };
}

function generateInvoicePdf(id) {
  const orders = getSheetData("Order");
  const o = orders.find(x => x['ID PESANAN'] === id);
  if (!o) return null;
  const fin = calculateOrderFinanceServer(o);
  const pdf = SpreadsheetApp.create('Invoice');
  const sheet = pdf.getActiveSheet();
  sheet.getRange('A1').setValue('LENTERA CENDEKIA');
  sheet.getRange('A2').setValue('Premium Academic Support');
  sheet.getRange('G1').setValue('INVOICE');
  sheet.getRange('G2').setValue('#' + id);
  sheet.getRange('A4').setValue('Diterbitkan Untuk:');
  sheet.getRange('A5').setValue(o['NAMA PELANGGAN']);
  sheet.getRange('A6').setValue('WA: ' + o['NO HANDPHONE']);
  sheet.getRange('G4').setValue('Status Bayar:');
  sheet.getRange('G5').setValue(fin.isPaidFull ? 'Paid Full' : (o['STATUS PEMBAYARAN'] || 'Unpaid'));
  sheet.getRange('G6').setValue(new Date().toLocaleDateString('id-ID', {day:'numeric', month:'long', year:'numeric'}));
  sheet.getRange('A8').setValue('Deskripsi');
  sheet.getRange('F8').setValue('Qty');
  sheet.getRange('G8').setValue('Total');
  sheet.getRange('A9').setValue(o['LAYANAN']);
  sheet.getRange('A10').setValue(o['RINCIAN PESANAN'] || 'Pengerjaan akademik kualitas premium.');
  sheet.getRange('F9').setValue((o.QTY || 1) + ' ' + (o.SATUAN || 'Unit'));
  sheet.getRange('G9').setValue(fin.gross);
  sheet.getRange('G11').setValue('Diskon: -' + fin.gross * fin.discountP / 100);
  sheet.getRange('G12').setValue('Net: ' + fin.net);
  sheet.getRange('A12').setValue('Riwayat Pembayaran:');
  let historyRow = 13;
  if (fin.originalDpAmount > 0) {
    sheet.getRange('A' + historyRow).setValue('DP (' + fin.dpP + '%)');
    sheet.getRange('C' + historyRow).setValue(fin.originalDpAmount);
    historyRow++;
  }
  if (fin.isPaidFull) {
    sheet.getRange('A' + historyRow).setValue('Pelunasan');
    sheet.getRange('C' + historyRow).setValue(fin.pelunasanAmount > 0 ? fin.pelunasanAmount : fin.net);
    historyRow++;
  }

  sheet.getRange('G13').setValue('DP Paid: -' + fin.dpAmount);
  sheet.getRange('G14').setValue('SISA TAGIHAN: ' + fin.balance);
  const url = 'https://docs.google.com/spreadsheets/d/' + pdf.getId() + '/export?exportFormat=pdf';
  const response = UrlFetchApp.fetch(url, { headers: { authorization: 'Bearer ' + ScriptApp.getOAuthToken() } });
  const blob = response.getBlob().setName('LC-Invoice-' + id + '.pdf');
  DriveApp.getFileById(pdf.getId()).setTrashed(true);  
  return blob;
}

function sendWhatsApp(phone, message, orderId = "", mediaUrl = "") {
  const TOKEN = "DKYLem6wuRH9U1xTmQt2";
  let cleanPhone = String(phone).replace(/\D/g, '');
  if (cleanPhone.startsWith('0')) cleanPhone = '62' + cleanPhone.substring(1);
  if (!cleanPhone.startsWith('62')) cleanPhone = '62' + cleanPhone;

  const url = "https://api.fonnte.com/send";
  const payload = { "target": cleanPhone, "message": message, "url": mediaUrl || "", "filename": mediaUrl ? "QRCode-Pembayaran.png" : "" };
  const options = { "method": "post", "headers": { "Authorization": TOKEN }, "contentType": "application/json", "payload": JSON.stringify(payload), "muteHttpExceptions": true };

  try {
    const response = UrlFetchApp.fetch(url, options);
    const result = JSON.parse(response.getContentText());
    if (result.status === true) { logWA(orderId, cleanPhone, "Pengiriman WA", message, "Sukses"); return true; } 
    else { logWA(orderId, cleanPhone, "Pengiriman WA", message, "Gagal", result.message); return false; }
  } catch (e) { logWA(orderId, cleanPhone, "Pengiriman WA", message, "Gagal", e.toString()); return false; }
}

  }

  const fin = calculateOrderFinanceServer(order);
  const type = String(mode || '').trim().toUpperCase();
  const qrisUrl = "https://ibb.co.com/cSksQpdd";
  const qrisViewUrl = "https://ibb.co.com/cSksQpdd";

  let nominal = 0;
  let label = "";
  if (type === "DP") {
    nominal = fin.originalDpAmount > 0 ? fin.originalDpAmount : (fin.net * 0.5);

  } else {
    nominal = fin.balance > 0 ? fin.balance : Math.max(0, fin.net - fin.originalDpAmount);

    `Konfirmasi setelah transfer ya 🙏`,
    ``,
    `Lentera Cendekia`
  ].join("\n");

  const ok = sendWhatsApp(order['NO HANDPHONE'], message, orderId, qrisUrl);
  return {
    success: ok,

function getNotifications(username, role) {
  const lastSeenKey = 'lastSeen_' + username;
  let lastSeen = PropertiesService.getScriptProperties().getProperty(lastSeenKey);
  if (!lastSeen) lastSeen = '2020-01-01T00:00:00Z';
  const lastSeenDate = new Date(lastSeen);
  const orders = getSheetData("Order");
  const notifications = [];

  orders.forEach(o => {
    if (!o['WAKTU  INPUT PESANAN']) return;
    const orderTime = new Date(o['WAKTU  INPUT PESANAN']);
    if (orderTime <= lastSeenDate) return;

    let message = ""; let type = "";
    const isAdmin = String(role).toLowerCase() === 'admin';
    const isWorker = String(role).toLowerCase() === 'worker';

    if (o['TAHAPAN PESANAN'] === "Antrean" && !o.WORKER) { if (isAdmin || isWorker) { message = `Pesanan baru #${o['ID PESANAN']} dari ${o['NAMA PELANGGAN']}`; type = "new_order"; } } 
    else if (o['TAHAPAN PESANAN'] === "Selesai") { message = `Pesanan #${o['ID PESANAN']} telah selesai`; type = "completed"; } 
    else if (o['TAHAPAN PESANAN'] === "Proses" || o['TAHAPAN PESANAN'] === "Revisi") { message = `Status pesanan #${o['ID PESANAN']} diubah menjadi ${o['TAHAPAN PESANAN']}`; type = "status_update"; }

    if (message) notifications.push({ id: o['ID PESANAN'], time: orderTime.toISOString(), message: message, type: type, orderId: o['ID PESANAN'] });
  });

  notifications.sort((a, b) => new Date(b.time) - new Date(a.time));
  return notifications.slice(0, 30);
}

function ensurePengeluaranSumberDanaColumn_() {
  const sheet = SHEETS.PENGELUARAN;
  const lastCol = Math.max(sheet.getLastColumn(), 1);
  const headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(h => String(h || '').trim());
  if (!headers.includes('Sumber_dana')) {
    sheet.getRange(1, lastCol + 1).setValue('Sumber_dana').setFontWeight('bold').setHorizontalAlignment('center');
    return lastCol + 1;
  }
  return headers.indexOf('Sumber_dana') + 1;
}

function normalizeSumberDana_(sources) {
  const aliases = {
    'gaji': 'Sumber dana gaji',
    'sumber dana gaji': 'Sumber dana gaji',
    'pendapatan lainnya': 'Pendapatan lainnya',
    'pendapatan lain': 'Pendapatan lainnya',
    'kas perusahaan': 'Kas perusahaan'
  };
  const raw = Array.isArray(sources) ? sources : String(sources || '').split(/[|,;]/);
  const out = [];
  raw.forEach(item => {
    const key = String(item || '').trim().toLowerCase();
    const normalized = aliases[key];
    if (normalized && !out.includes(normalized)) out.push(normalized);
  });
  return out;
}

function getSumberDanaAvailability(excludeExpenseId) {
  const orders = getSheetData('Order').filter(o => String(o['TAHAPAN PESANAN'] || '').trim() === 'Selesai');
  const services = getSheetData('Layanan');
  let totalDanaGaji = 0;
  let totalKasPerusahaan = 0;
  orders.forEach(o => {
    const fin = calculateOrderFinanceServer(o);
    const layanan = String(o['LAYANAN'] || '').trim().toLowerCase();
    const svc = services.find(s => String(s['Detail Layanan'] || '').trim().toLowerCase() === layanan) || {};
    const feeCompanyPct = Number(svc['fee perusahaan(%)']) || 0;
    const feeCompany = (Number(fin.net) || 0) * feeCompanyPct / 100;
    totalKasPerusahaan += feeCompany;
    totalDanaGaji += Math.max(0, (Number(fin.net) || 0) - feeCompany);
  });
  const totalPendapatanLain = getSheetData('Pendapatan').reduce((sum, p) => sum + (Number(p['Jumlah']) || 0), 0);
  const expenses = getSheetData('Pengeluaran').filter(p => String(p['ID Pengeluaran'] || '') !== String(excludeExpenseId || ''));
  const used = { 'Sumber dana gaji': 0, 'Pendapatan lainnya': 0, 'Kas perusahaan': 0 };

  expenses.forEach(p => {
    const jumlah = Number(p['Jumlah']) || 0;
    const sources = normalizeSumberDana_(p['Sumber_dana'] || p['Sumber Dana'] || 'Kas perusahaan');
    const activeSources = sources.length ? sources : ['Kas perusahaan'];
    const perSource = jumlah / activeSources.length;
    activeSources.forEach(src => {
      used[src] = (used[src] || 0) + perSource;
    });
  });

  const pools = {
    'Sumber dana gaji': totalDanaGaji,
    'Pendapatan lainnya': totalPendapatanLain,
    'Kas perusahaan': totalKasPerusahaan
  };

  return Object.keys(pools).map(name => ({
    name: name,
    total: pools[name],
    used: used[name] || 0,
    available: Math.max(0, pools[name] - (used[name] || 0))
  }));
}

function validateSumberDana_(payload) {
  const jumlah = Number(payload.jumlah) || 0;
  const sources = normalizeSumberDana_(payload.sumberDana || payload.Sumber_dana || payload['Sumber_dana']);
  if (jumlah <= 0) return { success: false, message: 'Jumlah pengeluaran harus lebih dari 0.' };
  if (!sources.length) return { success: false, message: 'Pilih minimal satu sumber dana.' };

  const availability = getSumberDanaAvailability(payload.id);
  const totalAvailable = availability
    .filter(item => sources.includes(item.name))
    .reduce((sum, item) => sum + (Number(item.available) || 0), 0);

  if (jumlah > totalAvailable) {
    return {
      success: false,
      message: 'Dana tidak cukup. Total tersedia dari sumber terpilih: Rp ' + totalAvailable.toLocaleString('id-ID'),
      availability: availability
    };
  }
  return { success: true, sources: sources, availability: availability };
}

function deleteRelatedGaji(deskripsiPengeluaran, jumlahPengeluaran) {
  const gajiSheet = SHEETS.GAJI;
  const gData = gajiSheet.getDataRange().getValues();
  
  for (let i = 1; i < gData.length; i++) {
    let bulanTahunRaw = gData[i][2];                    // Kolom Bulan/Tahun
    const totalTerima = Number(gData[i][6]) || 0;      // Kolom Total Terima

    // Parse jika berupa Date atau ISO String
    let bulanTahunText = "";
    if (bulanTahunRaw instanceof Date) {
      bulanTahunText = Utilities.formatDate(bulanTahunRaw, "GMT+7", "MMMM yyyy");
    } else if (typeof bulanTahunRaw === 'string' && bulanTahunRaw.includes('T')) {
      const dateObj = new Date(bulanTahunRaw);
      if (!isNaN(dateObj)) {
        bulanTahunText = Utilities.formatDate(dateObj, "GMT+7", "MMMM yyyy");
      }
    } else {
      bulanTahunText = String(bulanTahunRaw || "").trim();
    }

    const jumlahCocok = Math.abs(totalTerima - jumlahPengeluaran) < 10000;
    const bulanCocok = deskripsiPengeluaran.toLowerCase().includes(bulanTahunText.toLowerCase());

    if (jumlahCocok && bulanCocok) {
      gajiSheet.deleteRow(i + 1);
      console.log(`✅ Gaji berhasil dihapus otomatis: ${bulanTahunText}`);
      return true;
    }
  }
  
  console.log("⚠️ Tidak menemukan record Gaji yang cocok untuk dihapus");
  return false;
}

}

// ==================== CLAIM ORDER - VERSI ROBUST ====================
function claimOrder(orderId, username) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    const sheet = ss.getSheetByName("Order");
    if (!sheet) return { success: false, message: "Sheet Order tidak ditemukan" };

    const data = sheet.getDataRange().getValues();
    const headers = data[0];

    // Cari kolom dengan tepat
    let workerCol = -1;        // Kolom P → WORKER (nama worker)
    let timestampCol = -1;     // Kolom U → WAKTU DITERIMA WORKER

    for (let i = 0; i < headers.length; i++) {
      const h = String(headers[i]).trim().toUpperCase();
      if (h === "WORKER") {
        if (workerCol === -1) workerCol = i + 1;      // Kolom pertama = P
        else timestampCol = i + 1;                    // Kolom kedua = U
      }
    }

    if (workerCol === -1) return { success: false, message: "Kolom WORKER (P) tidak ditemukan" };

    // Cari baris order
    let rowToUpdate = -1;
    for (let i = 1; i < data.length; i++) {
      if (String(data[i][0]).trim() === String(orderId).trim()) {
        rowToUpdate = i + 1;
        break;
      }
    }

    if (rowToUpdate === -1) return { success: false, message: "Order tidak ditemukan" };

    // Cek sudah di-claim atau belum
    if (data[rowToUpdate - 1][workerCol - 1]) {
      return { success: false, message: "Order sudah diambil orang lain" };
    }

    const now = new Date();

    // Isi kolom P (nama worker)
    sheet.getRange(rowToUpdate, workerCol).setValue(username);

    // Isi kolom U (waktu diterima worker)
    if (timestampCol !== -1) {
      sheet.getRange(rowToUpdate, timestampCol).setValue(now);
    }

    return { success: true, message: "Berhasil mengambil pekerjaan" };

  } catch (e) {
    console.error("Error claimOrder:", e);
    return { success: false, message: "Terjadi kesalahan sistem" };
  }
}

// ==================== BULK UPDATE STATUS LAYANAN ====================
function bulkUpdateServiceStatus(payload) {
  const { ids, status } = payload;
  if (!ids || !Array.isArray(ids) || ids.length === 0) {
    return { success: false, message: "Tidak ada ID layanan yang dipilih" };
  }
  
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("LAYANAN") || 
                SpreadsheetApp.getActiveSpreadsheet().getSheetByName("SERVICES") || 
                SpreadsheetApp.getActiveSpreadsheet().getSheets()[0];
  
  const data = sheet.getDataRange().getValues();
  const headers = data[0];
  
  let idCol = -1, statusCol = -1;
  
  for (let i = 0; i < headers.length; i++) {
    if (headers[i] === "ID LAYANAN" || headers[i] === "ID") idCol = i;
    if (headers[i] === "Status Layanan" || headers[i] === "Status") statusCol = i;
  }
  
  if (idCol === -1 || statusCol === -1) {
    return { success: false, message: "Kolom ID LAYANAN atau Status Layanan tidak ditemukan" };
  }
  
  let updated = 0;
  
  for (let i = 1; i < data.length; i++) {
    const rowId = String(data[i][idCol]);
    if (ids.includes(rowId)) {
      sheet.getRange(i + 1, statusCol + 1).setValue(status);
      updated++;
    }
  }
  
  return { 
    success: true, 
    message: `${updated} layanan berhasil diupdate menjadi "${status}"` 
  };
}
